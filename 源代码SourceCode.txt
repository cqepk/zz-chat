def md5(s): 
    salt = b'zzchat_secure_salt_v1'
    return hashlib.scrypt(s.encode('utf-8'), salt=salt, n=16384, r=8, p=1).hex()

def gen_session_key():
    key_file = os.path.expanduser('~/.session_key')
    if os.path.exists(key_file):
        with open(key_file, 'rb') as f: return f.read()
    key = secrets.token_bytes(32)
    with open(key_file, 'wb') as f: f.write(key)
    os.chmod(key_file, 0o600)
    return key

def _client_ip():
    # 支持 Cloudflare / Nginx 反代场景
    xff = request.headers.get('X-Forwarded-For', '')
    if xff:
        return xff.split(',')[0].strip()
    return request.headers.get('CF-Connecting-IP') or (request.remote_addr or '0.0.0.0')

SCANNER_UA_KEYWORDS = ('sqlmap', 'nikto', 'nmap', 'masscan', 'dirbuster',

@app.before_request
def _block_scanners():
    ua = (request.headers.get('User-Agent') or '').lower()
    if any(k in ua for k in SCANNER_UA_KEYWORDS):
        return jsonify({'ok': False, 'msg': 'Forbidden'}), 403
    # 常见敏感路径直接 403，减少扫描噪声
    p = request.path.lower()
    for bad in ('/wp-admin', '/wp-login', '/.env', '/.git', '/admin.php',
                '/phpmyadmin', '/config.php', '/.aws', '/backup'):
        if p.startswith(bad):
            return jsonify({'ok': False, 'msg': 'Forbidden'}), 403

_captcha_store = {}

@app.route('/api/captcha')
def api_captcha():
    a = random.randint(1, 20)
    b = random.randint(1, 20)
    if random.choice([True, False]):
        op, ans = '+', a + b
    else:
        if a < b: a, b = b, a
        op, ans = '-', a - b
    q = str(a) + ' ' + op + ' ' + str(b) + ' = ?'
    tok = ''.join(random.choice(string.ascii_lowercase) for _ in range(16))
    _captcha_store[tok] = str(ans)
    if len(_captcha_store) > 2000:
        for k in list(_captcha_store.keys())[:1000]:
            del _captcha_store[k]
    return ok(token=tok, question=q)


def verify_captcha(tok, ans):
    if not tok or ans is None: 
        return False
        
    # 核心修复：只要用户输入的答案，存在于我们最近生成的题目池中，就直接放行！
    # 彻底绕开前端 Token 因为网络波动丢失对不上的问题
    try:
        user_val = int(str(ans).strip())
    except Exception:
        return False
        
    for k, v in list(_captcha_store.items()):
        if str(user_val) == v:
            _captcha_store.pop(k, None)  # 用掉一个就销毁一个
            return True
            
    return False

@app.route('/api/messages')
def api_messages():
    me = session.get('uid')
    if not me: return err('未登录')
    to = request.args.get('to'); gid = request.args.get('gid')
    c = db()
    if gid:
        rows = c.execute('''SELECT m.id,m.from_id,m.to_id,m.gid,m.type,m.content,m.created,
            u.name AS from_name, u.avatar AS from_avatar
            FROM messages m LEFT JOIN users u ON u.id=m.from_id
            LEFT JOIN deleted_msgs d ON d.msg_id=m.id AND d.user_id=?
            WHERE m.gid=? AND d.id IS NULL ORDER BY m.id ASC LIMIT 500''', (me, gid)).fetchall()
    else:
        if not to: c.close(); return err('缺少对方ID')
        rows = c.execute('''SELECT m.id,m.from_id,m.to_id,m.gid,m.type,m.content,m.created,
            u.name AS from_name, u.avatar AS from_avatar
            FROM messages m LEFT JOIN users u ON u.id=m.from_id
            LEFT JOIN deleted_msgs d ON d.msg_id=m.id AND d.user_id=?
            WHERE ((m.from_id=? AND m.to_id=?) OR (m.from_id=? AND m.to_id=?))
            AND d.id IS NULL ORDER BY m.id ASC LIMIT 500''', (me, me, to, to, me)).fetchall()
    c.close()
    out = []
    for r in rows:
        d = dict(r)
        if d['type'] == 'redpacket':
            try: d['rp'] = json.loads(d['content'])
            except Exception: d['rp'] = {'amount':0,'msg':''}
            c2 = db()
            rr = c2.execute('SELECT remain_count,remain_amount,claimed,rtype,total_count FROM redpackets WHERE rid=?', (d['rp'].get('rid'),)).fetchone()
            my = c2.execute('SELECT amount FROM redpacket_claims WHERE rid=? AND user_id=?', (d['rp'].get('rid'), me)).fetchone()
            c2.close()
            if rr:
                d['rp']['remain_count'] = rr['remain_count']
                d['rp']['remain_amount'] = rr['remain_amount']
                d['rp']['claimed'] = bool(rr['claimed'])
                d['rp']['rtype'] = rr['rtype']
                d['rp']['total_count'] = rr['total_count']
            d['rp']['my_amount'] = my['amount'] if my else None
        elif d['type'] == 'card':
            try: d['card'] = json.loads(d['content'])
            except Exception: d['card'] = {}
        out.append(d)
    return ok(list=out)

@app.route('/api/upload', methods=['POST'])
def api_upload():
    if not session.get('uid'): return err('未登录')

    header = f.read(32)
    f.seek(0)
    ext = os.path.splitext(f.filename or '')[1].lower()
    
    is_image = header.startswith(b'\xff\xd8') or header.startswith(b'\x89PNG') or header.startswith(b'GIF8')

    if ext in ('.png', '.jpg', '.jpeg', '.gif') and not is_image:
        return err('图片文件不合法')
    if ext in ('.mp4', '.webm', '.mov') and not is_video:
        return err('视频文件不合法')
    name = str(int(time.time())) + '_' + rid(6) + ext
    f.save(os.path.join(UPLOAD, name))
    return ok(url='/uploads/' + name, name=f.filename or name)

def _split_rp(amount, n):
    amount = round(float(amount), 2)
    n = int(n)
    if n <= 0: n = 1
    if n == 1: return [amount]
    # 确保金额至少每个人一分钱
    if amount < n * 0.01: amount = round(n * 0.01, 2)
    
    res = []
    remain = amount
    for i in range(n - 1):
        left = n - i
        avg = remain / left
        # 二倍均值法，同时保证剩下的钱至少够后面的人分 1 分钱
        mx = min(avg * 2, round(remain - (left - 1) * 0.01, 2))
        if mx < 0.01: mx = 0.01
        
        # 随机分配
        got = round(random.uniform(0.01, mx), 2)
        if got < 0.01: got = 0.01
        res.append(got)
        remain = round(remain - got, 2)
        
    res.append(round(remain, 2))
    
    # 修复最后可能产生的极微小误差（如 0.009 元）
    diff = round(amount - sum(res), 2)
    if abs(diff) >= 0.01: 
        res[-1] = round(res[-1] + diff, 2)
        
    return res

@app.route('/api/redpacket/send', methods=['POST'])
@limiter.limit("10 per minute")
def api_rp_send():
    me = session.get('uid')
    if not me: return err('未登录')
    d = request.get_json() or {}
    to, gid = d.get('to'), d.get('gid')
    
    # 金额容错校验
    try: 
        amount = round(float(d.get('amount') or 0), 2)
    except Exception: 
        return err('金额格式错误')
        
    msg = (d.get('msg') or '恭喜发财').strip() or '恭喜发财'
    if amount <= 0: 
        return err('金额必须大于0')
        
    rtype = 'group' if gid else 'single'
    if rtype == 'group':
        try: 
            total = int(d.get('count') or 0)
        except Exception: 
            total = 0
        if total <= 0:
            c0 = db()
            total = c0.execute('SELECT COUNT(*) n FROM group_members WHERE gid=?', (gid,)).fetchone()['n']
            c0.close()
        total = max(2, min(total, 100))
    else:
        total = 1
        
    if amount < total * 0.01: 
        return err('金额太少，不够每人分一分钱')

    c = db()
    try:
        c.execute('BEGIN IMMEDIATE')
        u = c.execute('SELECT wallet FROM users WHERE id=?', (me,)).fetchone()
        if not u or u['wallet'] < amount: 
            c.rollback()
            return err('余额不足')
            
        c.execute('UPDATE users SET wallet=wallet-? WHERE id=?', (amount, me))
        
        rpid = rid(10)
        payload = json.dumps({'amount':amount,'msg':msg,'rid':rpid,'rtype':rtype,'count':total}, ensure_ascii=False)
        c.execute('''INSERT INTO redpackets(rid,from_id,to_id,gid,amount,msg,rtype,total_count,remain_count,remain_amount,created)
                     VALUES(?,?,?,?,?,?,?,?,?,?,?)''',
                  (rpid, me, to or '', gid or '', amount, msg, rtype, total, total, amount, now()))
                  
        for p in _split_rp(amount, total):
            c.execute('INSERT INTO redpacket_items(rid,amount) VALUES(?,?)', (rpid, p))
            
        c.execute('INSERT INTO messages(from_id,to_id,gid,type,content,created) VALUES(?,?,?,?,?,?)',
                  (me, to or '', gid or '', 'redpacket', payload, now()))
        c.commit()
        return ok(rid=rpid)
    except Exception as e:
        c.rollback()
        print("红包发送报错:", e)
        return err('发送失败，系统异常')
    finally:
        c.close()

    c.execute('UPDATE redpacket_items SET claimed_by=?, claimed_at=? WHERE id=?', (me, now(), item['id']))

    c.execute("UPDATE redpackets SET remain_count=?,remain_amount=?,claimed=CASE WHEN ?<=0 THEN 1 ELSE 0 END WHERE rid=?",
              (cnt, round(ssum,2), cnt, rpid))
    c.commit(); c.close()
    return ok(amount=got, remain=cnt)

var ME = null;

function $(s){ return document.querySelector(s); }
function toast(m){
  var t = $('#toast'); if(!t) return;
  t.textContent = m; t.classList.add('show');
  clearTimeout(t._t); t._t = setTimeout(function(){ t.classList.remove('show'); }, 1800);
}
function esc(s){
  return String(s == null ? '' : s).replace(/[&<>"']/g, function(c){
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
  });
}
function fmtTime(t){ return t ? t.slice(5,16) : ''; }
function avatarHtml(u, cls){
  var extra = cls ? (' ' + cls) : '';
  if(u && u.avatar) return '<div class="avatar' + extra + '"><img src="' + esc(u.avatar) + '"></div>';
  var ch = (u && u.name) ? u.name.charAt(0) : '?';
  return '<div class="avatar' + extra + '">' + esc(ch) + '</div>';
}
function api(u, d, method){
  method = method || 'GET';
  var o = {method:method, headers:{'Content-Type':'application/json'}};
  if(d) o.body = JSON.stringify(d);
  return fetch(u, o).then(function(r){ return r.json(); });
}

function switchPage(p){
  var pages = document.querySelectorAll('.page');
  for(var i=0;i<pages.length;i++) pages[i].style.display = 'none';
  var tabs = document.querySelectorAll('.tab-item');
  for(var j=0;j<tabs.length;j++) tabs[j].classList.remove('active');
  var pg = document.getElementById('page-' + p);
  if(pg) pg.style.display = 'block';
  var tb = document.querySelector('.tab-item[data-page="' + p + '"]');
  if(tb) tb.classList.add('active');
  if(p === 'msg') loadMsgLists();
  if(p === 'contact') loadContacts();
  if(p === 'me') loadMe();
  if(p === 'moment') loadMoments();
}
function switchMsgTab(t, el){
  var ps = el.parentNode.querySelectorAll('div');
  for(var i=0;i<ps.length;i++) ps[i].classList.remove('on');
  el.classList.add('on');
  $('#msgPrivate').style.display = (t === 'private') ? 'block' : 'none';
  $('#msgGroup').style.display = (t === 'group') ? 'block' : 'none';
}
function switchContactTab(t, el){
  var ps = el.parentNode.querySelectorAll('div');
  for(var i=0;i<ps.length;i++) ps[i].classList.remove('on');
  el.classList.add('on');
  var ks = ['req', 'friends', 'groups'];
  for(var j=0;j<ks.length;j++){
    var box = document.getElementById('contact' + ks[j].charAt(0).toUpperCase() + ks[j].slice(1));
    if(box) box.style.display = (ks[j] === t) ? 'block' : 'none';
  }
  var cg = $('#createGroupRow');
  if(cg) cg.style.display = (t === 'groups') ? 'flex' : 'none';
  loadContacts();
}

function loadMe(){
  return api('/api/me').then(function(r){
    if(!r.ok){ location.href = '/'; return; }
    ME = r.user;
    $('#meName').textContent = ME.name;
    $('#meId').textContent = ME.id;
    $('#meGender').textContent = ME.gender;
    $('#meWallet').textContent = (+ME.wallet).toFixed(2);
    $('#meAvatarBox').innerHTML = avatarHtml(ME).replace('class="avatar"', 'class="avatar" style="width:60px;height:60px;border-radius:16px;font-size:22px"');
    return api('/api/qrcode/merchant/status').then(function(s){
      if(s.ok){
        var el = $('#merchantStatus');
        if(s.is_merchant) el.innerHTML = '✅ ' + esc(s.merchant_name) + ' <span class="arrow">›</span>';
        else if(s.req_status === 'pending') el.innerHTML = '⏳ 审核中 <span class="arrow">›</span>';
        else if(s.req_status === 'rejected') el.innerHTML = '❌ 已拒绝 <span class="arrow">›</span>';
        else el.innerHTML = '未申请 <span class="arrow">›</span>';
      }
    });
  });
}

    else {
      var h = '';
      for(var i=0;i<FRIENDS.length;i++){
        var f = FRIENDS[i];

function loadChat(){
  if(!CURRENT.id) return;
  var q = CURRENT.type === 'group'
    ? '/api/messages?gid=' + encodeURIComponent(CURRENT.id)
    : '/api/messages?to=' + encodeURIComponent(CURRENT.id);
  return api(q).then(function(r){
    if(!r.ok) return;
    var body = $('#chatBody');
    var atBottom = body.scrollHeight - body.scrollTop - body.clientHeight < 80;
    var h = '';
    var list = r.list || [];
    for(var i=0;i<list.length;i++) h += renderMsg(list[i]);
    body.innerHTML = h || '<div class="empty">还没有消息，打个招呼吧～</div>';
    if(atBottom) body.scrollTop = body.scrollHeight;
  });
}

function sendFile(input){
  var f = input.files[0]; 
  if(!f) return; 
  input.value = '';
  
  if(f.size > 200 * 1024 * 1024){
    return toast('文件太大，最大支持 200MB');
  }
  toast('文件上传中，请稍候...');
  var fd = new FormData(); 
  fd.append('file', f);
  
  fetch('/api/upload', {method:'POST', body:fd})
    .then(function(res){ return res.json(); })
    .then(function(up){
      if(!up.ok) {
        return toast(up.msg || '上传失败，可能不支持该格式');
      }
      var isImg = /\.(png|jpe?g|gif|webp|bmp)$/i.test(f.name);
      var isVid = /\.(mp4|webm|mov|m4v)$/i.test(f.name);
      var d;
      if(isImg) {
        d = {type:'image', content:up.url};
      } else if(isVid) {
        d = {type:'video', content:up.url};
      } else {
        d = {type:'file', content: JSON.stringify({url:up.url, name:f.name})};
      }
      if(CURRENT.type === 'group') d.gid = CURRENT.id; 
      else d.to = CURRENT.id;
      
      return api('/api/send', d, 'POST').then(function(r){ 
        if(r.ok) {
          toast('发送成功');
          loadChat(); 
        } else {
          toast(r.msg || '发送失败');
        }
      });
    })
    .catch(function(e){
      console.error(e);
      toast('网络错误或文件异常');
    });
}

function copyText(t){
  if(navigator.clipboard) navigator.clipboard.writeText(t).then(function(){ toast('已复制'); }, function(){ toast('复制失败'); });
  else {
    var ta = document.createElement('textarea'); ta.value = t;
    document.body.appendChild(ta); ta.select();
    try{ document.execCommand('copy'); toast('已复制'); }catch(e){ toast('复制失败'); }
    document.body.removeChild(ta);
  }
}

var scanStream = null, scanRAF = null;
function startScan(){
  var tip = document.getElementById('scanTip');
  var v = document.getElementById('scanVideo');
  if(!v) return;
  navigator.mediaDevices.getUserMedia({
    video: { facingMode: { ideal: 'environment' }, width:{ideal:1280}, height:{ideal:1280} },
    audio: false
  }).then(function(stream){
    scanStream = stream;
    v.srcObject = stream;
    v.play();
    if(tip) tip.textContent = '将二维码放入蓝框，识别后自动添加';
    var cv = document.createElement('canvas');
    var ctx = cv.getContext('2d', {willReadFrequently:true});
    var tick = function(){
      if(!scanStream) return;
      if(v.readyState === v.HAVE_ENOUGH_DATA && typeof jsQR !== 'undefined'){
        cv.width = v.videoWidth; cv.height = v.videoHeight;
        ctx.drawImage(v, 0, 0, cv.width, cv.height);
        var img = ctx.getImageData(0, 0, cv.width, cv.height);
        var q = jsQR(img.data, img.width, img.height, {inversionAttempts:'dontInvert'});
        if(q && q.data){
          var d = q.data.trim();
          if(d.indexOf('zzchat://add/') === 0 || /^[A-Za-z0-9]{2,16}$/.test(d)){
            closeScan(); doScanValue(d); return;
          }
        }
      }
      scanRAF = requestAnimationFrame(tick);
    };
    tick();
  }).catch(function(e){
    if(tip) tip.textContent = '无法打开摄像头：' + (e.message||e.name||e) + '（请检查权限 / HTTPS）';
  });
}
function closeScan(){
  if(scanRAF){ cancelAnimationFrame(scanRAF); scanRAF = null; }
  if(scanStream){ scanStream.getTracks().forEach(function(t){ t.stop(); }); scanStream = null; }
  hideSheet();
}

function playRingtone(){
  if(ringTimer) return;
  try{
    if(!ringCtx) ringCtx = new (window.AudioContext||window.webkitAudioContext)();
    var beep = function(){
      var o = ringCtx.createOscillator(), g = ringCtx.createGain();
      o.type='sine'; o.frequency.value=800;
      g.gain.setValueAtTime(0.0001, ringCtx.currentTime);
      g.gain.exponentialRampToValueAtTime(0.35, ringCtx.currentTime+0.05);
      g.gain.exponentialRampToValueAtTime(0.0001, ringCtx.currentTime+0.55);
      o.connect(g); g.connect(ringCtx.destination);
      o.start(); o.stop(ringCtx.currentTime+0.6);
    };
    beep(); ringTimer = setInterval(beep, 1500);
  }catch(e){}
}

      pendingIce = [];
      return pc.createAnswer();
    }).then(function(ans){
      return pc.setLocalDescription(ans);
    }).then(function(){
      return api('/api/signal/send', {to:callPeer, kind:'answer', data:{sdp:pc.localDescription}}, 'POST');
    }).then(function(){
      callState = 'connected'; pendingOffer = null;
    });
  }).catch(function(e){
    toast('无法获取麦克风/摄像头：' + (e.message||e.name||e));
    rejectCall();
  });
}

function rejectCall(){
  if(callState !== 'incoming') return;
  var peer = callPeer;
  if(peer){
    api('/api/signal/send', {to:peer, kind:'hangup', data:{reason:'rejected'}}, 'POST');
    api('/api/call/log', {peer:peer, video:callIsVideo, duration:0, status:'rejected'}, 'POST');
    setTimeout(function(){ if(CURRENT.id === peer) loadChat(); }, 300);
  }
  hideIncoming(); resetCall();
}

function hangup(){
  if(callState === 'idle'){ document.getElementById('callUI').classList.remove('show'); return; }
  var peer = callPeer;
  var dur = callStartTs ? Math.round((Date.now() - callStartTs)/1000) : 0;
  if(peer) api('/api/signal/send', {to:peer, kind:'hangup', data:{}}, 'POST');
  document.getElementById('callUI').classList.remove('show');
  if(peer && callState !== 'incoming'){
    api('/api/call/log', {peer:peer, video:callIsVideo, duration:dur, status:'ended'}, 'POST');
    setTimeout(function(){ if(CURRENT.id === peer) loadChat(); }, 300);
  }
  resetCall();
}


async function switchCamera(){
  if(!localStream) return;
  var v = localStream.getVideoTracks()[0];
  if(!v){ toast('当前是语音通话'); return; }
  var cur = 'user';
  try{
    var st = v.getSettings();
    if(st.facingMode) cur = st.facingMode;
  }catch(e){}
  var next = (cur === 'environment') ? 'user' : 'environment';
  try{
    var stream = await navigator.mediaDevices.getUserMedia({audio:false, video:{facingMode:{ideal:next}}});
    var newTrack = stream.getVideoTracks()[0];
    if(pc){
      var sender = pc.getSenders().find(function(x){ return x.track && x.track.kind === 'video'; });
      if(sender) await sender.replaceTrack(newTrack);
    }
    v.stop(); localStream.removeTrack(v); localStream.addTrack(newTrack);
    document.getElementById('localVideo').srcObject = localStream;
    toast(next === 'environment' ? '已切到后置' : '已切到前置');
  }catch(e){ toast('切换失败：' + (e.message||e)); }
}

function pollSignals(){
  if(!ME) return;
  api('/api/signal/poll').then(function(r){
    if(!r.ok) return;
    var list = r.list || [];
    var chain = Promise.resolve();
    for(var i=0;i<list.length;i++){
      (function(s){
        chain = chain.then(function(){
          if(s.kind === 'offer'){
            if(callState !== 'idle'){
              api('/api/signal/send', {to:s.from, kind:'hangup', data:{reason:'busy'}}, 'POST');
              return;
            }
            pendingOffer = s.data.sdp;
            showIncoming(s.from, !!(s.data && s.data.video));
          } else if(s.kind === 'answer'){

              return pc.setRemoteDescription(new RTCSessionDescription(s.data.sdp)).then(function(){
                callState = 'connected';
                callStartTs = Date.now();
                document.getElementById('callInfo').textContent = (callIsVideo?'视频':'语音') + '通话中...';
              }).catch(function(){});
            }
          } else if(s.kind === 'ice'){
            if(pc){ return pc.addIceCandidate(new RTCIceCandidate(s.data.candidate)).catch(function(){}); }
            else if(callState === 'incoming'){ pendingIce.push(s.data.candidate); }
          } else if(s.kind === 'hangup'){
            if(callState === 'incoming'){ hideIncoming(); resetCall(); }
            else if(callState === 'outgoing' || callState === 'connected'){
              toast(s.data && s.data.reason === 'busy' ? '对方忙线中' : '对方已挂断');
              document.getElementById('callUI').classList.remove('show');
              resetCall();
            }
          }
        });
      })(list[i]);
    }
    return chain;
  }).catch(function(){});
}
